# Simulation parameters
sigma_epsilon <- 1
p_negative_theta <- 0.2
mu_theta <- 1

n_subjects <- 85
n_trials <- 200

# Derived parameters
sigma_theta <- -mu_theta / qnorm(p_negative_theta)
id <- rep(1:n_subjects, each = n_trials)

# Simulate data
set.seed(123)
theta <- rnorm(n_subjects, mean = mu_theta, sd = sigma_theta)
y <- rnorm(n_subjects * n_trials, mean = theta[id], sd = sigma_epsilon)
sim_data <- data.frame(
  id = id,
  y = y
)

# Simulate data

sigma_epsilon <- 1
mu_theta <- 1.2
sigma_theta <- 1

n_subjects <- 85
n_trials <- 200
id <- rep(1:n_subjects, each = n_trials)

set.seed(123)
theta <- rlnorm(n_subjects, meanlog = mu_theta, sdlog = sigma_theta)
y <- rnorm(n_subjects*n_trials, mean = theta[id], sd = sigma_epsilon)

sim_data_logN <- data.frame(id = id, y = y)

# Visualize data

theta |> as.data.frame() |>
  ggplot(aes(x = theta)) +
  geom_histogram(bins = 15, color = "black", fill = "lightblue") +
  labs(title = "Histogram of Theta", x = "Theta", y = "Count")
sim_data_logN |>
  dplyr::summarize(
    mean_y = mean(y)
    , .by = id
  ) |>
  ggplot(aes(x = mean_y)) +

  labs(title = "Histogram of Subject Means", x = "Mean Y", y = "Count")
# Testing model
